    const missing = [];
    if (!formData.clientId)        missing.push("ID Cliente");
    if (!formData.nombre)          missing.push("Nombre Completo");
    if (!formData.tipoIdentificacion) missing.push("Tipo de Identificación");

}
    if (formData.estudiosPoligrafia) {

      ...formData,
      visitaDomiciliaria:  formData.visitaDomiciliaria  ? 'SI' : 'NO',

      referenciacion:      formData.referenciacion      ? 'SI' : 'NO',

        const aliasName = `${newId}_${att.file.name}`;
        await uploadFileViaProxy(att.file, aliasName);
        await runServer('registerTempDocument', { requestId: newId, docName: att.name, fileName: aliasName });
        processed++;
        setUploadProgress(p => ({ ...p, current: processed }));
      }

      setCreatedId(newId);

            <span className="text-[10px] font-black text-gray-400 uppercase tracking-widest">Evaluado</span>
            <span className="font-black text-gray-700 truncate ml-4 uppercase text-[11px]">{formData.nombre}</span>
          </div>

        <div className="grid grid-cols-1 gap-4">
          <button onClick={resetForm} className="btn-3d-blue py-4 rounded-2xl flex items-center justify-center gap-3">
            <Icon name="ClipboardList" size={18}/> <span className="text-xs font-black uppercase tracking-widest">Crear Nueva</span>
          </button>
          <button onClick={() => setView('history')} className="btn-3d-gray py-4 rounded-2xl flex items-center justify-center gap-3">
            <Icon name="History" size={18}/> <span className="text-xs font-black uppercase tracking-widest text-gray-600">Ver Historial</span>
          </button>
        </div>
      </div>
    </div>

  const inputClass = "w-full border border-gray-200 rounded-xl p-2.5 text-xs focus:ring-2 focus:ring-vividBlue/20 focus:border-vividBlue outline-none transition-all bg-white hover:border-gray-300";
  const labelClass = "text-[10px] font-black text-gray-500 uppercase tracking-wider mb-1.5 block";
  const fieldClass = "flex flex-col";

  return (
    <div className="p-4 md:p-8 max-w-5xl mx-auto animate-fade-in pb-24">

      <div className="flex items-center justify-center mb-10">
        <div className="flex items-center group cursor-pointer" onClick={() => setStep(1)}>
          <div className={`w-10 h-10 rounded-2xl flex items-center justify-center font-black text-sm transition-all shadow-lg ${step===1?'bg-vividRed text-white shadow-red-500/40 scale-110':'bg-white text-gray-400 border border-gray-100'}`}>1</div>

          <div className="ml-4">
            <span className={`block text-[10px] font-black uppercase tracking-widest ${step===2?'text-vividRed':'text-gray-400'}`}>Paso 2</span>

              Fecha: {formData.fechaSolicitud.split(',')[0]}
            </div>
          </div>

          {/* ── TARJETA ÚNICA ── */}
          <div className="bg-white rounded-2xl border border-gray-200 shadow-sm divide-y divide-gray-100">

          {/* SECCIÓN 1 */}
          <div className="p-6">
          <FormSection icon="Building" title="1. Información del Cliente">
            <div className="grid grid-cols-1 md:grid-cols-3 gap-5">

const BulkUpload = ({ userContext, setView, onReload }) => {
  const [selectedClient, setSelectedClient] = useState('');
  const [downloading, setDownloading]       = useState(false);
  const [uploading, setUploading]           = useState(false);
  const [file, setFile]                     = useState(null);
  const [uploadResult, setUploadResult]     = useState(null);
  const [showValidationAlert, setShowValidationAlert] = useState(false);
  const [validationErrors, setValidationErrors]       = useState([]);

  const clients = useMemo(() => userContext?.allowedClientIds || [], [userContext]);
  useEffect(() => { if (clients.length > 0 && !selectedClient) setSelectedClient(clients[0]); }, [clients]);

  const handleDownloadTemplate = async () => {
    if (!selectedClient) { alert("Por favor seleccione un cliente primero."); return; }
    setDownloading(true);
    try {
      const response = await runServer('getTemplateName', { clientId: selectedClient });
      if (!response.success) throw new Error(response.message || "Error al obtener plantilla.");
      const iframe = document.createElement('iframe');
      iframe.style.display = "none";
      iframe.src = `${PROXY_URL}?filename=${encodeURIComponent(response.filename)}&token=${PROXY_TOKEN}`;
      document.body.appendChild(iframe);
      setTimeout(() => { setDownloading(false); setTimeout(() => { if(document.body.contains(iframe)) document.body.removeChild(iframe); }, 60000); }, 3000);
    } catch (e) { alert("Error: " + e.message); setDownloading(false); }
  };

  const handleUpload = async () => {
    if (!selectedClient) return alert("Seleccione un cliente.");
    if (!file) return alert("Seleccione un archivo CSV para cargar.");
    setUploading(true);
    const reader = new FileReader();
    reader.onload = async (e) => {
      try {
        const csvContent = e.target.result;
        const base64Content = csvContent.split(',')[1];
        const decodedText = atob(base64Content);
        const rows = decodedText.split(/\r\n|\n|\r/);
        if (rows.length < 2) throw new Error("El archivo está vacío o sin datos.");
        const delimiter = rows[0].includes(';') ? ';' : ',';
        const headers = rows[0].split(delimiter).map(h => h.trim().replace(/^"|"$/g, '').toUpperCase());
        const correoIdx  = headers.findIndex(h => h.includes("CORREO") || h.includes("EMAIL"));
        const celularIdx = headers.findIndex(h => h.includes("CELULAR") || h.includes("MOVIL") || h.includes("TELEFONO"));
        const frontErrors = [];
        for (let i = 1; i < rows.length; i++) {
          const line = rows[i].trim(); if (!line) continue;
          const cols = line.split(delimiter); const rowNum = i + 1;
          if (correoIdx !== -1 && cols[correoIdx]) {
            const val = cols[correoIdx].trim().replace(/^"|"$/g, '');
            if (val && !isValidEmail(val)) frontErrors.push(`Fila ${rowNum}: Correo inválido (${val})`);
          }
          if (celularIdx !== -1 && cols[celularIdx]) {
            const val = cols[celularIdx].trim().replace(/^"|"$/g, '');
            if (val && !isValidPhone(val)) frontErrors.push(`Fila ${rowNum}: Celular inválido (${val})`);
          }
        }
        if (frontErrors.length > 0) { setValidationErrors(frontErrors); setShowValidationAlert(true); setUploading(false); return; }
        const response = await runServer('processBulkUpload', { csvContent: base64Content, clientId: selectedClient });
        if (response.success) { setUploadResult(response); if (onReload) onReload(); }
        else {
          setValidationErrors(response.errorList || [response.message]);
          setShowValidationAlert(true);
        }
      } catch (err) { alert("Error en carga masiva: " + err.message); }
      finally { setUploading(false); }
    };
    reader.readAsDataURL(file);
  };

  if (uploadResult) return (
    <div className="fixed inset-0 z-[100] flex items-center justify-center p-6 bg-black/30 backdrop-blur-sm animate-fade-in">
      <div className="bg-white p-10 rounded-3xl shadow-2xl text-center max-w-md w-full border border-gray-100 animate-scale-in">
        <div className="w-24 h-24 bg-vividGreen/10 rounded-3xl flex items-center justify-center mx-auto mb-8 shadow-inner transform rotate-12">
          <Icon name="CheckCircle" size={48} className="text-vividGreen -rotate-12"/>
        </div>
        <h2 className="text-3xl font-black text-gray-800 mb-2 tracking-tight">¡Carga Finalizada!</h2>
        <p className="text-gray-400 text-xs font-bold uppercase tracking-widest mb-8 text-center">Resultados del procesamiento masivo</p>
        <div className="bg-gray-50/50 p-6 rounded-3xl border border-gray-100 text-left mb-6 space-y-4">
          <div className="flex justify-between items-center">
            <span className="text-[10px] font-black text-gray-400 uppercase tracking-widest">Exitosos</span>
            <span className="font-mono font-black text-vividGreen bg-white px-3 py-1 rounded-xl shadow-sm border border-gray-100">{uploadResult.loaded}</span>
          </div>
          <div className="flex justify-between items-center">
            <span className="text-[10px] font-black text-gray-400 uppercase tracking-widest">Fallidos</span>
            <span className={`font-mono font-black px-3 py-1 rounded-xl shadow-sm border border-gray-100 ${uploadResult.failed > 0 ? 'text-vividRed bg-vividRed/5' : 'text-gray-400 bg-white'}`}>{uploadResult.failed}</span>
          </div>
        </div>
        {uploadResult.technicalErrors?.length > 0 && (
          <div className="bg-vividRed/[0.03] border border-vividRed/10 rounded-2xl p-4 mb-8 text-left max-h-36 overflow-y-auto">
            <p className="text-[9px] font-black text-vividRed uppercase tracking-widest mb-2">Errores Técnicos Detectados:</p>
            <ul className="text-[10px] text-gray-600 space-y-1.5 list-disc list-inside font-medium">
              {uploadResult.technicalErrors.map((e,i) => <li key={i} className="leading-tight">{e}</li>)}
            </ul>
          </div>
        )}
        <div className="grid grid-cols-1 gap-4">
          <button onClick={() => { setFile(null); setUploadResult(null); }} className="btn-3d-blue py-4 rounded-2xl flex items-center justify-center gap-3">
            <Icon name="Upload" size={18}/> <span className="text-xs font-black uppercase tracking-widest">Subir Otro</span>
          </button>
          <button onClick={() => setView('history')} className="btn-3d-gray py-4 rounded-2xl flex items-center justify-center gap-3">
            <Icon name="History" size={18}/> <span className="text-xs font-black uppercase tracking-widest text-gray-600">Ver Historial</span>
          </button>
        </div>
      </div>
    </div>
  );

  return (
    <div className="p-6 md:p-8 max-w-4xl mx-auto animate-fade-in">
      <h2 className="text-2xl font-bold text-gray-800 mb-6">Carga Masiva</h2>
      <div className="bg-white p-8 rounded-xl shadow-sm border border-gray-200">
        <div className="mb-8 max-w-xs"><label className="block text-xs font-bold text-gray-500 uppercase mb-1">Cliente</label>
          <select value={selectedClient} onChange={e => setSelectedClient(e.target.value)} className="w-full border border-gray-300 rounded-lg p-2 text-sm focus:ring-1 focus:ring-appBlue outline-none">
            <option value="">Seleccione...</option>
            {clients.map(id => <option key={id} value={id}>{userContext?.clientNames?.[id] || id}</option>)}
          </select>
        </div>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
          <div className="border-r border-gray-100 pr-6">
            <div className="w-16 h-16 bg-blue-50 text-blue-600 rounded-full flex items-center justify-center mx-auto mb-4"><Icon name="Download" size={32}/></div>
            <h3 className="text-lg font-bold text-gray-700 text-center mb-2">Paso 1: Descargar Plantilla</h3>
            <p className="text-sm text-gray-500 text-center mb-5">Obtenga el formato CSV para este cliente.</p>
            <button onClick={handleDownloadTemplate} disabled={downloading || !selectedClient}
              className="flex items-center justify-center gap-2 mx-auto text-blue-600 font-bold hover:underline text-sm btn-action disabled:opacity-50">
              {downloading ? <><Icon name="RefreshCw" size={15} className="animate-spin"/> Buscando...</> : <><Icon name="Download" size={15}/> Descargar Plantilla</>}
            </button>
          </div>
          <div className="pl-0 md:pl-4">
            <div className="w-16 h-16 bg-green-50 text-green-600 rounded-full flex items-center justify-center mx-auto mb-4"><Icon name="Upload" size={32}/></div>
            <h3 className="text-lg font-bold text-gray-700 text-center mb-2">Paso 2: Subir Datos</h3>
            <p className="text-sm text-gray-500 text-center mb-4">Suba el CSV diligenciado.</p>
            <div className="border-2 border-dashed border-gray-300 rounded-xl p-4 hover:bg-gray-50 cursor-pointer transition relative mb-4">
              <input type="file" accept=".csv" onChange={e => e.target.files?.[0] && setFile(e.target.files[0])} className="absolute inset-0 w-full h-full opacity-0 cursor-pointer"/>
              <div className="text-center text-xs text-gray-400">
                {file ? <span className="text-green-600 font-bold">{file.name}</span> : "Arrastre o haga clic aquí"}
              </div>
            </div>
            <button onClick={handleUpload} disabled={uploading || !file}
              className="w-full btn-3d-red px-6 py-3 rounded-2xl font-black text-[10px] uppercase tracking-widest disabled:opacity-50 flex items-center justify-center gap-3">
              {uploading ? <><Icon name="RefreshCw" size={16} className="animate-spin"/> Procesando...</> : <><Icon name="Upload" size={16}/> Procesar Archivo</>}
            </button>
          </div>
        </div>
      </div>
      <AlertModal show={showValidationAlert} title="Errores en Plantilla"
        message="Se encontraron errores en el archivo. La carga fue rechazada."
        items={validationErrors} onClose={() => { setShowValidationAlert(false); setValidationErrors([]); }}/>
    </div>
  );
};

                <p className="text-xs text-gray-500">Configure restricciones por cliente.</p>
              </div>
            </div>
            <div className="relative w-full md:w-64">
              <div className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400"><Icon name="Search" size={14}/></div>
              <input type="text" placeholder="Buscar cliente..." value={clientSearch} onChange={e => setClientSearch(e.target.value)}
                className="w-full pl-9 pr-4 py-2 bg-gray-50 border border-gray-200 rounded-xl text-xs focus:ring-2 focus:ring-appRed/20 focus:border-appRed outline-none transition-all"/>
            </div>
          </div>
          <div className="max-h-96 overflow-y-auto border border-gray-100 rounded-xl shadow-inner bg-gray-50/30">
            <table className="w-full text-left text-xs">
              <thead className="bg-gray-50 text-gray-400 font-bold uppercase tracking-wider border-b border-gray-100">
                <tr><th className="px-4 py-3">Cliente</th></tr>
              </thead>
              <tbody className="divide-y divide-gray-50">
                {filteredClientIds.length === 0 ? (
                  <tr><td className="px-4 py-8 text-center text-gray-400 italic">No se encontraron clientes.</td></tr>
                ) : filteredClientIds.map(id => (
                  <tr key={id} className="hover:bg-gray-50 transition-colors">
                    <td className="px-4 py-3 cursor-pointer" onClick={() => setSelectedAdminClient(id)}>
                      <div className="font-bold text-gray-700 hover:text-appBlue transition-colors flex items-center justify-between">
                        <div className="flex flex-col">
                          <span className="text-sm">{userContext?.clientNames?.[id] || id}</span>
                          <span className="text-[10px] text-gray-400 font-mono">{id}</span>
                        </div>
                        <Icon name="ChevronRight" size={16} className="text-gray-300"/>
                      </div>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      )}
      <div className="bg-white p-6 rounded-xl shadow-sm border border-gray-200 mb-6">
        <h3 className="font-bold text-gray-700 mb-4">Estado de Datos</h3>
        <div className="space-y-3">
          {[
            { label:'Solicitudes (Período Activo)', sub:'Carga filtrada por fecha.', syncing: syncStatus==='requests' },
            { label:'Tablas Maestras (Ciudades, Líneas CC...)', sub:'Datos de referencia para formularios.', syncing: syncStatus==='master' },
          ].map((item, i) => (
            <div key={i} className="flex items-center justify-between p-3 bg-gray-50 rounded-lg border border-gray-100">
              <div className="flex items-center gap-3">
                <div className={`p-2 rounded-full ${item.syncing ? 'bg-blue-100 text-blue-600' : 'bg-green-100 text-green-600'}`}>
                  {item.syncing ? <Icon name="RefreshCw" size={17} className="animate-spin"/> : <Icon name="CheckCircle" size={17}/>}
                </div>
                <div>
                  <div className="font-bold text-gray-700 text-sm">{item.label}</div>
                  <div className="text-xs text-gray-500">{item.sub}</div>
                </div>
              </div>
              <span className="text-xs font-bold text-gray-400">{lastSync ? 'Actualizado' : 'Pendiente'}</span>
            </div>
          ))}
          <div className="flex items-center justify-between p-3 bg-gray-50 rounded-lg border border-gray-100">
            <div className="flex items-center gap-3">
              <div className="p-2 rounded-full bg-gray-100 text-gray-600"><Icon name="User" size={17}/></div>
              <div>
                <div className="font-bold text-gray-700 text-sm">Clientes Asignados</div>
                <div className="text-xs text-gray-500">Actualiza tu lista sin recargar la página.</div>
              </div>
            </div>
            <button onClick={onRefreshClients} className="px-3 py-1.5 bg-white border rounded-lg text-xs font-bold text-gray-600 hover:bg-gray-100 transition btn-action">Actualizar</button>
          </div>
        </div>
      </div>
      <div className="bg-white p-6 rounded-xl shadow-sm border border-gray-200">
        <div className="flex items-center gap-4">
          <div className="h-14 w-14 bg-blue-100 rounded-full flex items-center justify-center text-blue-600"><Icon name="User" size={28}/></div>
          <div>
            <div className="text-xs uppercase font-bold text-gray-500">Cuenta Activa</div>
            <div className="text-lg font-bold text-gray-800">{userContext?.email}</div>
            <div className="inline-flex items-center gap-1 bg-green-100 text-green-700 px-2 py-0.5 rounded text-xs font-bold mt-1">
              <Icon name="ShieldCheck" size={11}/> {userContext?.role}
            </div>
          </div>
        </div>
      </div>
      {selectedAdminClient && (
        <AdminClientModal clientId={selectedAdminClient} clientName={userContext.clientNames[selectedAdminClient]}
          onClose={() => setSelectedAdminClient(null)} showToast={showToast}/>
      )}
    </div>
  );
};

// ─── APP PRINCIPAL ────────────────────────────────────────────────────────
const App = () => {
  const [userCtx,       setUserCtx]       = useState(null);
  const [initError,     setInitError]     = useState(null);
  const [sidebarOpen,   setSidebarOpen]   = useState(false);
  const [isSidebarCollapsed, setIsSidebarCollapsed] = useState(true);
  const [view,          setView]          = useState('loading');
  const [selectedRequest, setSelectedRequest] = useState(null);
  const [syncStatus,    setSyncStatus]    = useState('idle');
  const [lastSync,      setLastSync]      = useState(null);
  const [toast,         setToast]         = useState(null);
  const [initialData,   setInitialData]   = useState({ data: [], total: 0 });
  const [masterData,    setMasterData]    = useState(null);
  const [period,        setPeriod]        = useState('today');
  const [isLoading,     setIsLoading]     = useState(true);

  const showToast = (message, type = 'info') => setToast({ message, type });

  const loadRequests = useCallback(async (periodKey = period) => {
    if (syncStatus !== 'idle') return;
    setSyncStatus('requests');
    try {
      const reqs = await runServer('getRequests', { period: periodKey });
      setInitialData(reqs);
      setLastSync(Date.now());
    } catch (e) {
      showToast("Error cargando solicitudes: " + e.message, 'error');
    } finally {
      setSyncStatus('idle');
    }
  }, [period, syncStatus]);

  const loadMasterData = useCallback(async () => {
    setSyncStatus('master');
    try {
      const master = await runServer('getMasterData', {});
      setMasterData(master);
    } catch (e) {
      console.warn("Error cargando datos maestros:", e.message);
    } finally {
      setSyncStatus('idle');
    }
  }, []);

  const refreshAll = async (manual = false, periodKey = period) => {
    if (syncStatus !== 'idle' && !manual) return;
    try {
      setSyncStatus('requests');
      const reqs = await runServer('getRequests', { period: periodKey });
      setInitialData(reqs);
      setSyncStatus('master');
      const master = await runServer('getMasterData', {});
      setMasterData(master);
      setLastSync(Date.now());
      if (manual) showToast("Datos actualizados.", 'success');
    } catch (e) {
      showToast("Error en sincronización: " + e.message, 'error');
    } finally {
      setSyncStatus('idle');
    }
  };

  const handleChangePeriod = (newPeriod) => {
    setPeriod(newPeriod);
    loadRequests(newPeriod);
  };

  const handleRefreshClients = async () => {
    showToast("Actualizando clientes...", 'info');
    try {
      const updatedCtx = await runServer('refreshUserContext', {});
      setUserCtx(updatedCtx);
      await refreshAll(true, period);
      showToast("Clientes y datos actualizados.", 'success');
    } catch (e) {
      showToast("Error al actualizar: " + e.message, 'error');
    }
  };

  const handleLogin = async () => {
    setIsLoading(true);
    try {
      const ctx = await runServer('getUserContext', {});
      setUserCtx(ctx);
      if (ctx.isValidUser) {
        const isHistoryDefault = ['Cliente Consulta', 'Cliente Perfilado'].includes(ctx.role);
        const defaultPeriod = isHistoryDefault ? 'year' : 'today';
        setPeriod(defaultPeriod);
        setView(ctx.role === 'Cliente Consulta' ? 'history' : 'home');
        Promise.all([
          runServer('getRequests', { period: defaultPeriod }).then(r => { setInitialData(r); }),
          runServer('getMasterData', {}).then(m => { setMasterData(m); })
        ]).then(() => setLastSync(Date.now())).catch(console.warn);
      } else {
        setView('unauthorized');
      }
    } catch (e) {
      setInitError(e.message);
    } finally {
      setIsLoading(false);
    }
  };

  const handleLogout = useCallback(() => {
    setUserCtx(null); setInitError(null); setIsLoading(false); setView('loading');
  }, []);

  useEffect(() => {
    if (!userCtx) return;
    let timer;
    const resetTimer = () => {
